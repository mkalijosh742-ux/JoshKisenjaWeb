1. Introduction

1.1 Project Overview

The college requires an integrated database to manage academic and non-academic services: student records, course registration, teaching, assessments, accommodation, welfare, workload allocation, industry attachments and clubs. This document records the full design and implementation of the Student Services Management System (database: StudentServicesDB) using Microsoft SQL Server.

1.2 Objectives

  • Keep accurate, consistent records and minimise redundancy through normalisation to 3NF.
  • Enforce business rules (one course per student, assessment limits, 75% attendance, 60% pass mark) using constraints.
  • Provide role-based views for registrar, HODs, lecturers, dean, accounts and workload reporting.
  • Version-control all scripts on GitHub with annotated screenshots.

1.3 Key Business Rules

AreaRule
ModulesDiploma = 3, Certificate = 2, Artisan = 1; each module spans three terms.
SubjectsCode includes the module code; 2–6 teaching hours per week.
AssessmentCAT1 and CAT2 out of 20 each; exam out of 60. Exam needs ≥75% attendance and both CATs.
Pass markBelow 60% → supplementary exam after fee payment; results recorded separately.
EnrolmentOne course at a time; status: in session / on attachment / completed.
HostelAllocated only after confirmed accommodation payment.

5. Task 4 – DDL Operations

5.1 CREATE TABLE Statements with Constraints

-- Purpose: core schema with PK, FK, CHECK and UNIQUE constraints
CREATE TABLE Courses(
  CourseID INT IDENTITY PRIMARY KEY, CourseName VARCHAR(100) NOT NULL,
  Level VARCHAR(12) NOT NULL CHECK (Level IN ('Diploma','Certificate','Artisan')),
  DeptID INT NOT NULL REFERENCES Departments(DeptID));

CREATE TABLE Modules(
  ModuleCode VARCHAR(10) PRIMARY KEY, CourseID INT NOT NULL REFERENCES Courses(CourseID),
  ModuleNo TINYINT NOT NULL CHECK (ModuleNo BETWEEN 1 AND 3));

CREATE TABLE Subjects(
  SubjectCode VARCHAR(20) PRIMARY KEY,   -- e.g. M1-DBMS (module code prefix)
  SubjectName VARCHAR(100) NOT NULL,
  HoursPerWeek TINYINT NOT NULL CHECK (HoursPerWeek BETWEEN 2 AND 6),
  ModuleCode VARCHAR(10) NOT NULL REFERENCES Modules(ModuleCode));

CREATE TABLE Classes(
  ClassID INT IDENTITY PRIMARY KEY, ClassName VARCHAR(50) NOT NULL,
  CourseID INT NOT NULL REFERENCES Courses(CourseID), Intake VARCHAR(20) NOT NULL);

CREATE TABLE Students(
  StudentID INT IDENTITY PRIMARY KEY, AdmNo VARCHAR(20) NOT NULL UNIQUE,
  FullName VARCHAR(100) NOT NULL, DOB DATE NOT NULL,
  Gender CHAR(1) NOT NULL CHECK (Gender IN ('M','F')),
  Address VARCHAR(120), Phone VARCHAR(15),
  Status VARCHAR(15) NOT NULL DEFAULT 'in session'
        CHECK (Status IN ('in session','on attachment','completed')),
  CourseID INT NOT NULL REFERENCES Courses(CourseID),
  ClassID INT NOT NULL REFERENCES Classes(ClassID),
  CONSTRAINT UQ_Student_NameDOB UNIQUE (FullName, DOB));

CREATE TABLE Lecturers(
  LecturerID INT IDENTITY PRIMARY KEY, FullName VARCHAR(100) NOT NULL,
  Specialty VARCHAR(80), Role VARCHAR(40), Phone VARCHAR(15),
  DeptID INT NOT NULL REFERENCES Departments(DeptID));

CREATE TABLE Teaching(
  TeachID INT IDENTITY PRIMARY KEY,
  LecturerID INT NOT NULL REFERENCES Lecturers(LecturerID),
  SubjectCode VARCHAR(20) NOT NULL REFERENCES Subjects(SubjectCode),
  ClassID INT NOT NULL REFERENCES Classes(ClassID),
  Term TINYINT NOT NULL CHECK (Term BETWEEN 1 AND 3), Year INT NOT NULL, Hours TINYINT NOT NULL);

CREATE TABLE Assessments(
  AssessID INT IDENTITY PRIMARY KEY,
  StudentID INT NOT NULL REFERENCES Students(StudentID),
  SubjectCode VARCHAR(20) NOT NULL REFERENCES Subjects(SubjectCode),
  Term TINYINT NOT NULL, Year INT NOT NULL, Attendance DECIMAL(5,2),
  CAT1 DECIMAL(4,1) CHECK (CAT1 BETWEEN 0 AND 20),
  CAT2 DECIMAL(4,1) CHECK (CAT2 BETWEEN 0 AND 20),
  Exam DECIMAL(4,1) CHECK (Exam BETWEEN 0 AND 60),
  IsSupplementary BIT NOT NULL DEFAULT 0,
  UNIQUE (StudentID, SubjectCode, Term, Year, IsSupplementary));

CREATE TABLE Payments(
  PaymentID INT IDENTITY PRIMARY KEY, StudentID INT NOT NULL REFERENCES Students(StudentID),
  PayType VARCHAR(12) NOT NULL CHECK (PayType IN ('tuition','hostel','supp')),
  Amount DECIMAL(10,2) NOT NULL, Term TINYINT, Year INT, PaidOn DATE DEFAULT GETDATE());

CREATE TABLE Hostels(HostelID INT IDENTITY PRIMARY KEY, HostelName VARCHAR(50) NOT NULL, Capacity INT NOT NULL);
CREATE TABLE HostelAllocations(
  AllocID INT IDENTITY PRIMARY KEY, StudentID INT NOT NULL REFERENCES Students(StudentID),
  HostelID INT NOT NULL REFERENCES Hostels(HostelID), Room VARCHAR(10), Term TINYINT, Year INT);

CREATE TABLE Clubs(ClubID INT IDENTITY PRIMARY KEY, ClubName VARCHAR(60) NOT NULL UNIQUE,
  Patron VARCHAR(80), Chairperson VARCHAR(80), IsActive BIT DEFAULT 1);
CREATE TABLE ClubMembers(MemberID INT IDENTITY PRIMARY KEY,
  ClubID INT NOT NULL REFERENCES Clubs(ClubID), StudentID INT NOT NULL REFERENCES Students(StudentID),
  Year INT NOT NULL, UNIQUE (ClubID, StudentID, Year));

CREATE TABLE Firms(FirmID INT IDENTITY PRIMARY KEY, FirmName VARCHAR(100) NOT NULL);
CREATE TABLE Attachments(AttachID INT IDENTITY PRIMARY KEY,
  StudentID INT NOT NULL REFERENCES Students(StudentID), FirmID INT NOT NULL REFERENCES Firms(FirmID),
  StartDate DATE NOT NULL, EndDate DATE, AttachType VARCHAR(12) CHECK (AttachType IN ('attachment','internship')),
  Completed BIT DEFAULT 0);

6.5 Aggregate Queries

SELECT c.CourseName, COUNT(*) AS Total FROM Students s JOIN Courses c ON c.CourseID=s.CourseID GROUP BY c.CourseName;  -- (i)
SELECT d.DeptName, AVG(a.CAT1+a.CAT2+a.Exam) AS AvgMarks FROM Assessments a                                   -- (ii)
 JOIN Students s ON s.StudentID=a.StudentID JOIN Courses c ON c.CourseID=s.CourseID
 JOIN Departments d ON d.DeptID=c.DeptID WHERE a.Term=1 AND a.Year=2026 GROUP BY d.DeptName;
SELECT ClubID, Year, COUNT(*) AS Members FROM ClubMembers GROUP BY ClubID, Year;                              -- (iii)
SELECT SubjectCode, SUM(CAT1+CAT2+Exam) AS Total FROM Assessments                                             -- (iv)
 WHERE StudentID=1 AND Term=1 AND Year=2026 GROUP BY SubjectCode;
SELECT l.FullName, t.Term, SUM(t.Hours) AS HoursTaught FROM Teaching t                                        -- (ix)
 JOIN Lecturers l ON l.LecturerID=t.LecturerID GROUP BY l.FullName, t.Term;

7.1 Creating Views

ViewAudienceContent
vw_RegistrarRegistrarStudent records, course, class, status
vw_HODHODsDepartment students, lecturers, courses
vw_LecturerLecturersOwn classes and student marks
vw_DeanDeanInstitution-wide summaries
vw_ExamPerformanceExams officeMarks, totals, pass/fail
vw_AccountsAccountsPayments and fee balances
vw_WorkloadHODs / DeanLecturer hours per term
CREATE VIEW vw_ExamPerformance AS
SELECT a.StudentID, s.FullName, a.SubjectCode, a.Term, a.Year,
       a.CAT1+a.CAT2+a.Exam AS Total,
       CASE WHEN a.CAT1+a.CAT2+a.Exam >= 60 THEN 'Pass' ELSE 'Supplementary' END AS Result
FROM Assessments a JOIN Students s ON s.StudentID=a.StudentID
WHERE a.IsSupplementary = 0;
GO
CREATE VIEW vw_Workload AS
SELECT l.LecturerID, l.FullName, t.Term, t.Year, SUM(t.Hours) AS TotalHours
FROM Lecturers l JOIN Teaching t ON t.LecturerID=l.LecturerID GROUP BY l.LecturerID,l.FullName,t.Term,t.Year;

-- (iv) students in more than one club
SELECT StudentID, COUNT(*) AS Clubs FROM ClubMembers GROUP BY StudentID HAVING COUNT(*)>1 ORDER BY Clubs DESC;
